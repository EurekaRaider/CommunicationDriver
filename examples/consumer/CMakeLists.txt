cmake_minimum_required(VERSION 3.28)
project(CommunicationDriverConsumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CommunicationDriver CONFIG REQUIRED)
add_executable(consumer_c ../c/main.c)
target_link_libraries(consumer_c PRIVATE CommunicationDriver::communicationdriver)
add_executable(consumer_cpp ../cpp/main.cpp)
target_link_libraries(consumer_cpp PRIVATE CommunicationDriver::communicationdriver)
